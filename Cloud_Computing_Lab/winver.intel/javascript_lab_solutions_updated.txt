// Q1. [Category: Functions, Arrays, slice()] Write a JavaScript function to get the last element of an array. Passing parameter n will return the last n elements of the array.

function lastElements(arr, n = 1) {
    if (n === 1) {
        return arr[arr.length - 1];
    }

    return arr.slice(-n);
}

let arr = [10, 20, 30, 40, 50];

console.log(lastElements(arr));
console.log(lastElements(arr, 3));


// Q2. [Category: Recursion, Factorial] Write a recursive factorial function.

function factorial(n) {
    if (n < 0 || !Number.isInteger(n)) {
        return "Invalid input";
    }

    if (n === 0 || n === 1) {
        return 1;
    }

    return n * factorial(n - 1);
}

console.log(factorial(5));

// Q5. [Category: Arrays, Functions, Array Rotation] Implement a function that returns an updated array with r right rotations on an array of integers a.

function rightRotate(a, r) {
    if (a.length === 0) {
        return [];
    }

    r = r % a.length;

    if (r === 0) {
        return a.slice();
    }

    return a.slice(-r).concat(a.slice(0, a.length - r));
}

let numbers = [1, 2, 3, 4, 5];

console.log(rightRotate(numbers, 2));

// Q1. [Category: Arrays, Frequency Counting] Write a JavaScript program to find the most frequent item in an array.

let items = [1, 2, 3, 2, 4, 2, 3, 3, 3];
let frequency = {};
let mostFrequent;
let maxCount = 0;

for (let item of items) {
    frequency[item] = (frequency[item] || 0) + 1;

    if (frequency[item] > maxCount) {
        maxCount = frequency[item];
        mostFrequent = item;
    }
}
